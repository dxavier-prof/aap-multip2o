const $ = (selector) => document.querySelector(selector);

const aleatorio = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min;
const elegir    = (lista) => lista[aleatorio(0, lista.length - 1)];

/** Mezcla una copia del array (algoritmo Fisher-Yates) */
function barajar(lista){
  const a = [...lista];
  for(let i = a.length - 1; i > 0; i--){
    const j = aleatorio(0, i);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

/** Convierte segundos en "m:ss" */
function formatearTiempo(segundos){
  const m = Math.floor(segundos / 60);
  const s = Math.floor(segundos % 60);
  return `${m}:${String(s).padStart(2, '0')}`;
}

/* ============================================================
   3. GUARDADO EN localStorage
============================================================ */
const Progreso = {
  CLAVE: 'multipez:progreso:v1',
  datos: { estrellas: {}, nivelMaximo: null, sonido: true },

  cargar(){
    try{
      const guardado = localStorage.getItem(this.CLAVE);
      if(guardado) this.datos = Object.assign(this.datos, JSON.parse(guardado));
    }catch(err){ console.warn('No se pudo leer el guardado:', err); }
  },

  guardar(){
    try{ localStorage.setItem(this.CLAVE, JSON.stringify(this.datos)); }catch(err){}
  },

  /** Registra una victoria: conserva la mejor puntuación de estrellas */
  registrarVictoria(nivelId, estrellas){
    this.datos.estrellas[nivelId] = Math.max(this.datos.estrellas[nivelId] || 0, estrellas);
    const orden  = NIVELES.map(n => n.id);
    const actual = orden.indexOf(this.datos.nivelMaximo);   // -1 si no hay ninguno
    if(orden.indexOf(nivelId) > actual) this.datos.nivelMaximo = nivelId;
    this.guardar();
  },

  alternarSonido(){
    this.datos.sonido = !this.datos.sonido;
    this.guardar();
    return this.datos.sonido;
  }
};

/* ============================================================
   4. SONIDO CON WEB AUDIO API
   Efectos sintetizados: no hacen falta archivos externos.
   Para usar audios propios, basta con cargar <audio> o fetch()
   dentro de reproducir() (ver README, sección "Sonidos").
============================================================ */
const Sonido = (() => {
  let contexto = null;

  function obtenerContexto(){
    if(!contexto){
      const Ctx = window.AudioContext || window.webkitAudioContext;
      if(Ctx) contexto = new Ctx();
    }
    if(contexto && contexto.state === 'suspended') contexto.resume();
    return contexto;
  }

  /** Toca una nota corta con envolvente suave */
  function nota(frecuencia, duracion, tipo = 'sine', volumen = 0.15, retraso = 0){
    const ctx = obtenerContexto();
    if(!ctx || !Progreso.datos.sonido) return;
    const osc = ctx.createOscillator();
    const gan = ctx.createGain();
    osc.type = tipo;
    osc.frequency.value = frecuencia;
    const t0 = ctx.currentTime + retraso;
    gan.gain.setValueAtTime(0.0001, t0);
    gan.gain.exponentialRampToValueAtTime(volumen, t0 + 0.02);
    gan.gain.exponentialRampToValueAtTime(0.0001, t0 + duracion);
    osc.connect(gan);
    gan.connect(ctx.destination);
    osc.start(t0);
    osc.stop(t0 + duracion + 0.05);
  }

  return {
    reproducir(nombre){
      switch(nombre){
        case 'click':    nota(600, .08, 'triangle', .12); break;
        case 'tick':     nota(900, .04, 'square',  .05); break;
        case 'giro':     nota(500, .1,  'triangle', .08); break;
        case 'acierto':  [523, 659, 784].forEach((f, i) => nota(f, .15, 'triangle', .16, i * .09)); break;
        case 'error':    nota(220, .25, 'sawtooth', .12); nota(160, .3, 'sawtooth', .12, .12); break;
        case 'estrella': nota(880, .2, 'triangle', .18); nota(1175, .25, 'triangle', .16, .1); break;
        case 'victoria': [523, 587, 659, 784, 1047].forEach((f, i) => nota(f, .18, 'triangle', .16, i * .13)); break;
        case 'derrota':  [400, 340, 300, 240].forEach((f, i) => nota(f, .22, 'sine', .12, i * .16)); break;
      }
    }
  };
})();

/* ============================================================
   7. RULETA DE TABLAS
   Dibuja la ruleta en SVG y la hace girar con CSS.
============================================================ */
const NS_SVG = 'http://www.w3.org/2000/svg';

const Ruleta = {
  etiquetas: [],   // valores de los segmentos (números o '⭐')
  ticks: [],       // timeouts de sonidos programados

  /** Construye la ruleta según las tablas del nivel */
  construir(etiquetas){
    this.etiquetas = etiquetas;
    const svg = $('#ruleta-svg');
    while(svg.firstChild) svg.removeChild(svg.firstChild);

    const n = etiquetas.length, cx = 160, cy = 160, r = 150;
    const arco = 360 / n;

    // Segmentos + número de cada tabla
    etiquetas.forEach((etiqueta, i) => {
      const a0 = ((i * arco) - 90) * Math.PI / 180;
      const a1 = (((i + 1) * arco) - 90) * Math.PI / 180;
      const x0 = cx + r * Math.cos(a0), y0 = cy + r * Math.sin(a0);
      const x1 = cx + r * Math.cos(a1), y1 = cy + r * Math.sin(a1);

      const sector = document.createElementNS(NS_SVG, 'path');
      sector.setAttribute('d',
        `M ${cx} ${cy} L ${x0.toFixed(2)} ${y0.toFixed(2)} A ${r} ${r} 0 0 1 ${x1.toFixed(2)} ${y1.toFixed(2)} Z`);
      sector.setAttribute('fill', PALETA_RULETA[i % PALETA_RULETA.length]);
      sector.setAttribute('stroke', '#fffdf8');
      sector.setAttribute('stroke-width', '3');
      svg.appendChild(sector);

      const aM = ((i * arco + arco / 2) - 90) * Math.PI / 180;
      const texto = document.createElementNS(NS_SVG, 'text');
      texto.setAttribute('x', (cx + r * .65 * Math.cos(aM)).toFixed(1));
      texto.setAttribute('y', (cy + r * .65 * Math.sin(aM)).toFixed(1));
      texto.setAttribute('text-anchor', 'middle');
      texto.setAttribute('dominant-baseline', 'central');
      texto.setAttribute('class', 'ruleta-numero');
      texto.textContent = etiqueta;
      svg.appendChild(texto);
    });

    // Centro decorativo
    const centro = document.createElementNS(NS_SVG, 'circle');
    centro.setAttribute('cx', cx); centro.setAttribute('cy', cy); centro.setAttribute('r', '34');
    centro.setAttribute('fill', '#fffdf8');
    centro.setAttribute('stroke', '#ffd166'); centro.setAttribute('stroke-width', '5');
    svg.appendChild(centro);
    const emoji = document.createElementNS(NS_SVG, 'text');
    emoji.setAttribute('x', cx); emoji.setAttribute('y', cy + 1);
    emoji.setAttribute('text-anchor', 'middle'); emoji.setAttribute('dominant-baseline', 'central');
    emoji.setAttribute('font-size', '30'); emoji.textContent = '🎣';
    svg.appendChild(emoji);

    // Aro exterior y luces parpadeantes
    const aro = document.createElementNS(NS_SVG, 'circle');
    aro.setAttribute('cx', cx); aro.setAttribute('cy', cy); aro.setAttribute('r', '150');
    aro.setAttribute('fill', 'none'); aro.setAttribute('stroke', '#fffdf8'); aro.setAttribute('stroke-width', '6');
    svg.appendChild(aro);
    for(let i = 0; i < 12; i++){
      const aL = (i * 30 - 90) * Math.PI / 180;
      const luz = document.createElementNS(NS_SVG, 'circle');
      luz.setAttribute('cx', (cx + 150 * Math.cos(aL)).toFixed(1));
      luz.setAttribute('cy', (cy + 150 * Math.sin(aL)).toFixed(1));
      luz.setAttribute('r', '5'); luz.setAttribute('fill', '#fffdf8');
      luz.setAttribute('class', 'luz');
      luz.style.animationDelay = (i * .12) + 's';
      svg.appendChild(luz);
    }
  },

  /** Gira la ruleta hasta el segmento indicado y ejecuta el callback */
  girar(indice, alTerminar){
    const n = this.etiquetas.length, arco = 360 / n;
    const centroSegmento = indice * arco + arco / 2;
    const vueltas = aleatorio(5, 7);
    const margen  = (Math.random() * .6 - .3) * arco;   // no caer siempre en el borde
    const angulo  = vueltas * 360 + (360 - centroSegmento) + margen;

    const giro = $('#ruleta-giro');
    giro.style.transition  = 'none';
    giro.style.transform   = 'rotate(0deg)';
    void giro.getBoundingClientRect();   // fuerza reflujo para reiniciar la transición
    giro.style.transition  = 'transform 2.4s cubic-bezier(0.12, 0.75, 0.08, 1)';
    giro.style.transform   = `rotate(${angulo}deg)`;

    // "Ticks" mientras gira y aviso de fin
    this.limpiarTicks();
    let t = 120;
    for(let i = 0; i < 16; i++){
      t += 60 + i * i * 3.4;
      this.ticks.push(setTimeout(() => Sonido.reproducir('tick'), t));
    }
    this.ticks.push(setTimeout(alTerminar, 2550));
  },

  limpiarTicks(){
    this.ticks.forEach(clearTimeout);
    this.ticks = [];
  }
};

/* ============================================================
   8. LÓGICA PRINCIPAL DEL JUEGO
============================================================ */
const Juego = {
  nivel: null,
  ronda: 0,                 // problemas ya respondidos
  problema: null,
  vidas: VIDAS,
  correctas: 0,
  incorrectas: 0,
  racha: 0,
  mejorRacha: 0,
  respondido: false,
  rondasConCero: [],        // rondas donde la ruleta caerá en el 0 (regla garantizada)
  tiempoInicio: 0,
  cronometro: null,
  temporizador: null,
  timeoutRonda: null,

  /* ---------- Puesta en marcha de una partida ---------- */
  iniciar(nivelId){
    this.nivel = NIVELES.find(n => n.id === nivelId);
    this.ronda = 0; this.vidas = VIDAS;
    this.correctas = 0; this.incorrectas = 0;
    this.racha = 0; this.mejorRacha = 0;
    this.respondido = false; this.problema = null;

    // Garantizamos que la regla del 0 aparezca al menos 2 veces por nivel
    const sorteo = barajar([...Array(TOTAL_PROBLEMAS).keys()]);
    this.rondasConCero = [sorteo[0], sorteo[1]];

    // Preparamos la interfaz
    $('#chip-nivel').textContent = `${this.nivel.nombre} ${this.nivel.icono}`;
    $('#banner-feedback').hidden = true;
    pintarVidas();
    pintarContadores(this.correctas, this.incorrectas);
    pintarRed();
    $('#barra-ronda-fill').style.width = '0%';
    mostrarPantalla('pantalla-juego');

    // Cronómetro total de la partida
    this.tiempoInicio = Date.now();
    clearInterval(this.cronometro);
    this.actualizarCronometro();
    this.cronometro = setInterval(() => this.actualizarCronometro(), 500);

    this.abrirRuleta();
  },

  actualizarCronometro(){
    $('#cronometro').textContent = '⏱ ' + formatearTiempo((Date.now() - this.tiempoInicio) / 1000);
  },

  /* ---------- Ruleta (inicio de cada ronda) ---------- */
  abrirRuleta(){
    $('#banner-feedback').hidden = true;
    const etiquetas = [...this.nivel.tablas];
    if(this.nivel.dosCifras) etiquetas.push('⭐');
    Ruleta.construir(etiquetas);
    $('#ruleta-resultado').textContent = '¡Pulsa GIRAR para descubrir tu reto!';
    $('#btn-girar').disabled = false;
    $('#overlay-ruleta').hidden = false;
  },

  girarRuleta(){
    const btn = $('#btn-girar');
    if(btn.disabled) return;
    btn.disabled = true;
    Sonido.reproducir('giro');

    const etiquetas = Ruleta.etiquetas;
    let indice;
    if(this.rondasConCero.includes(this.ronda) && etiquetas.includes(0)){
      indice = etiquetas.indexOf(0);       // regla del 0 garantizada en estas rondas
    }else{
      indice = aleatorio(0, etiquetas.length - 1);
    }

    Ruleta.girar(indice, () => {
      const etiqueta = etiquetas[indice];
      let mensaje;
      if(etiqueta === '⭐')      mensaje = '⭐ ¡Reto de dos cifras!';
      else if(etiqueta === 0)    mensaje = '✨ ¡La regla mágica del 0!';
      else                       mensaje = `¡Tabla del ${etiqueta}!`;
      $('#ruleta-resultado').textContent = mensaje;
      Sonido.reproducir('estrella');
      setTimeout(() => {
        $('#overlay-ruleta').hidden = true;
        this.plantearProblema(etiqueta);
      }, 950);
    });
  },

  /* ---------- Generación del problema ---------- */
  plantearProblema(tabla){
    this.respondido = false;
    this.problema = this.crearProblema(tabla);
    const p = this.problema;

    $('#pregunta-texto').innerHTML =
      `¿Cuánto es <span class="num">${p.a}</span> <span class="signo">×</span> <span class="num">${p.b}</span>?`;
    $('#texto-ronda').textContent =
      `Problema ${Math.min(this.ronda + 1, TOTAL_PROBLEMAS)} / ${TOTAL_PROBLEMAS}`;

    this.pintarOpciones(this.generarOpciones(p.a, p.b));
    if(this.nivel.tiempoPorProblema > 0) this.iniciarTemporizador();
  },

  crearProblema(tabla){
    let a, b;
    if(tabla === '⭐'){                    // reto de dos cifras (solo Maestro)
      a = elegir([11, 12, 13, 15, 20, 25]);
      b = elegir([2, 3, 4, 5]);
    }else if(tabla === 0){                 // regla mágica: 0 × n
      a = 0; b = aleatorio(1, 10);
    }else{
      a = tabla; b = aleatorio(0, 10);     // b puede ser 0 → también aparece n × 0
    }
    // En Maestro mezclamos el orden a menudo; en el resto, de vez en cuando
    const probabilidadInvertir = this.nivel.dosCifras ? .5 : (tabla === 0 ? .5 : .2);
    if(Math.random() < probabilidadInvertir) [a, b] = [b, a];
    return { a, b, respuesta: a * b, esCero: a === 0 || b === 0 };
  },

  /** Genera 4 opciones: la correcta + 3 "trampas" plausibles */
  generarOpciones(a, b){
    const correcta = a * b;
    const opciones = new Set([correcta]);
    let candidatas;
    if(correcta === 0){
      // Trampas típicas con el 0: el propio número, la suma, el 10…
      candidatas = [a === 0 ? b : a, b === 0 ? a : b, a + b, 10, (a || b) + 1];
    }else{
      candidatas = [
        correcta + 1, correcta - 1, correcta + 2, correcta - 2,
        correcta + 10, correcta - 10,
        a * (b + 1), a * (b - 1), (a + 1) * b, (a - 1) * b, a + b
      ];
    }
    candidatas = barajar(candidatas.filter(n => Number.isInteger(n) && n >= 0 && n !== correcta));
    while(opciones.size < 4 && candidatas.length) opciones.add(candidatas.pop());
    let respaldo = correcta + 3;   // por si faltaran candidatas válidas
    while(opciones.size < 4){
      if(respaldo >= 0 && !opciones.has(respaldo)) opciones.add(respaldo);
      respaldo += aleatorio(2, 5);
    }
    return barajar([...opciones]);
  },

  /* ---------- Los 4 peces-respuesta ---------- */
  pintarOpciones(valores){
    const banco = $('#banco-peces');
    banco.innerHTML = '';
    const colores = ['#ffd166', '#74d3ae', '#84c7ef', '#f4989c'];

    valores.forEach((valor, i) => {
      const boton = document.createElement('button');
      boton.type = 'button';
      boton.className = 'pez';
      boton.style.setProperty('--pez-color', colores[i]);
      boton.style.animationDelay = (i * .35) + 's';
      boton.dataset.valor = valor;
      boton.setAttribute('aria-label', `Respuesta ${valor}`);
      boton.innerHTML = `
        <svg viewBox="0 0 130 86" aria-hidden="true">
          <path class="pez-cola" d="M28 43 L6 22 L13 43 L6 64 Z"></path>
          <ellipse class="pez-cuerpo" cx="72" cy="43" rx="50" ry="30"></ellipse>
          <path class="pez-aleta" d="M64 15 Q76 2 88 14 Q78 20 64 15 Z"></path>
          <circle class="pez-ojo-blanco" cx="101" cy="36" r="6.5"></circle>
          <circle class="pez-ojo" cx="102.5" cy="36" r="3"></circle>
          <path class="pez-boca" d="M114 48 Q118 51 114 55" fill="none"></path>
          <text class="pez-numero" x="66" y="45" text-anchor="middle" dominant-baseline="central">${valor}</text>
        </svg>`;
      boton.addEventListener('click', () => this.responder(valor, boton));
      banco.appendChild(boton);
    });
  },

  bloquearPeces(){
    document.querySelectorAll('.pez').forEach(p => p.disabled = true);
  },

  /* ---------- Respuesta del jugador ---------- */
  responder(valor, boton, porTiempo = false){
    if(this.respondido || !this.problema) return;
    this.respondido = true;
    this.detenerTemporizador();
    if(valor === this.problema.respuesta) this.acertar(boton);
    else this.fallar(boton, porTiempo);
  },

  acertar(boton){
    this.correctas++;
    this.racha++;
    this.mejorRacha = Math.max(this.mejorRacha, this.racha);
    Sonido.reproducir('acierto');
    boton.classList.add('pez-correcto');
    this.bloquearPeces();
    this.marcarRed(true);
    this.mostrarMensaje(true,
      `${elegir(FRASES_ACIERTO)}<br>
       <span class="dato">${this.problema.a} × ${this.problema.b} = ${this.problema.respuesta}</span>
       ${this.problema.esCero ? '<br><span class="regla">✨ ¡Todo número × 0 da 0!</span>' : ''}`);
    pintarContadores(this.correctas, this.incorrectas);
    this.avanzarTras(1900);
  },

  fallar(boton, porTiempo){
    this.incorrectas++;
    this.racha = 0;
    this.vidas--;
    Sonido.reproducir('error');
    if(boton) boton.classList.add('pez-incorrecto');
    this.bloquearPeces();

    // Revelamos cuál era el pez correcto
    const correcto = [...document.querySelectorAll('.pez')]
      .find(p => Number(p.dataset.valor) === this.problema.respuesta);
    if(correcto) correcto.classList.add('pez-revelado');

    this.marcarRed(false);
    pintarVidas();
    const aviso = porTiempo ? '⏰ ¡Se acabó el tiempo!' : elegir(FRASES_ERROR);
    this.mostrarMensaje(false,
      `${aviso}<br>
       <span class="dato">${this.problema.a} × ${this.problema.b} = ${this.problema.respuesta}</span>
       ${this.problema.esCero ? '<br><span class="regla">✨ Recuerda: ¡todo número × 0 da 0!</span>' : ''}`);
    pintarContadores(this.correctas, this.incorrectas);

    if(this.vidas <= 0) setTimeout(() => this.terminar(false), 2300);
    else this.avanzarTras(2300);
  },

  mostrarMensaje(esAcierto, html){
    const banner = $('#banner-feedback');
    banner.className = 'banner-feedback ' + (esAcierto ? 'ok' : 'mal');
    banner.innerHTML = html;
    banner.hidden = false;
  },

  marcarRed(acertado){
    const casilla = document.querySelectorAll('.red-casilla')[this.ronda];
    if(casilla){
      casilla.textContent = acertado ? '🐟' : '💦';
      casilla.classList.add(acertado ? 'casilla-pesca' : 'casilla-fallo');
    }
    $('#barra-ronda-fill').style.width = ((this.ronda + 1) / TOTAL_PROBLEMAS * 100) + '%';
  },

  avanzarTras(ms){
    clearTimeout(this.timeoutRonda);
    this.timeoutRonda = setTimeout(() => this.siguiente(), ms);
  },

  siguiente(){
    this.ronda++;
    if(this.ronda >= TOTAL_PROBLEMAS){ this.terminar(true); return; }
    this.abrirRuleta();
  },

  /* ---------- Temporizador por problema (solo Maestro) ---------- */
  iniciarTemporizador(){
    const barra = $('#temporizador');
    const fill  = $('#temporizador-fill');
    barra.hidden = false;
    fill.style.width = '100%';
    fill.classList.remove('urgente');

    const total = this.nivel.tiempoPorProblema * 1000;
    const inicio = Date.now();
    clearInterval(this.temporizador);
    this.temporizador = setInterval(() => {
      const restante = Math.max(0, 1 - (Date.now() - inicio) / total);
      fill.style.width = (restante * 100) + '%';
      if(restante < .3) fill.classList.add('urgente');
      if(restante <= 0){
        this.detenerTemporizador();
        this.responder(null, null, true);   // tiempo agotado = fallo
      }
    }, 100);
  },

  detenerTemporizador(){
    clearInterval(this.temporizador);
    this.temporizador = null;
    const barra = $('#temporizador');
    if(barra) barra.hidden = true;
  },

  /* ---------- Fin de partida ---------- */
  terminar(victoria){
    clearInterval(this.cronometro);
    this.detenerTemporizador();
    this.limpiarTimeouts();

    const segundos  = (Date.now() - this.tiempoInicio) / 1000;
    const precision = Math.round(this.correctas / TOTAL_PROBLEMAS * 100);
    let estrellas = 0;

    if(victoria){
      estrellas = this.correctas >= 20 ? 3 : this.correctas >= 19 ? 2 : 1;
      Progreso.registrarVictoria(this.nivel.id, estrellas);
      Sonido.reproducir('victoria');
      lanzarConfeti();
    }else{
      Sonido.reproducir('derrota');
    }

    pintarPantallaFinal({
      victoria, estrellas, segundos, precision, nivel: this.nivel,
      correctas: this.correctas, incorrectas: this.incorrectas,
      mejorRacha: this.mejorRacha
    });
    mostrarPantalla('pantalla-final');
  },

  limpiarTimeouts(){
    clearTimeout(this.timeoutRonda);
    Ruleta.limpiarTicks();
  }
};

/* ============================================================
   9. PANTALLA FINAL (estrellas, estadísticas y confeti)
============================================================ */
function pintarPantallaFinal(res){
  $('#final-emoji').textContent = res.victoria ? (res.estrellas === 3 ? '🏆' : '🎉') : '🌧️';
  $('#final-titulo').textContent = res.victoria ? '¡Nivel completado!' : '¡Oh no! Se acabaron las vidas';

  // Estrellas que aparecen una a una
  const cont = $('#estrellas-final');
  cont.innerHTML = '';
  for(let i = 0; i < 3; i++){
    const e = document.createElement('span');
    const ganada = i < res.estrellas;
    e.className = 'estrella' + (ganada ? ' ganada' : '');
    e.textContent = ganada ? '⭐' : '☆';
    if(ganada){
      e.style.animationDelay = (.35 + i * .45) + 's';
      setTimeout(() => Sonido.reproducir('estrella'), 400 + i * 450);
    }
    cont.appendChild(e);
  }

  // Mensaje motivador personalizado
  const mensaje = res.victoria ? elegir(MENSAJES_FINAL[res.estrellas]) : elegir(MENSAJES_DERROTA);
  $('#final-mensaje').innerHTML = res.victoria
    ? `${mensaje}<br><small>Progreso guardado: tu red de ${res.nivel.nombre} brilla con ${'★'.repeat(res.estrellas)} 🐚</small>`
    : mensaje;

  // Estadísticas de la partida
  $('#estadisticas').innerHTML = `
    <div class="stat"><span class="stat-icono">✅</span><span class="stat-valor">${res.correctas}</span><span class="stat-nombre">Correctas</span></div>
    <div class="stat"><span class="stat-icono">💦</span><span class="stat-valor">${res.incorrectas}</span><span class="stat-nombre">Fallidas</span></div>
    <div class="stat"><span class="stat-icono">⏱️</span><span class="stat-valor">${formatearTiempo(res.segundos)}</span><span class="stat-nombre">Tiempo</span></div>
    <div class="stat"><span class="stat-icono">🔥</span><span class="stat-valor">${res.mejorRacha}</span><span class="stat-nombre">Mejor racha</span></div>
    <div class="stat"><span class="stat-icono">🎯</span><span class="stat-valor">${res.precision}%</span><span class="stat-nombre">Precisión</span></div>`;

  // Botón de siguiente nivel (solo si tiene sentido)
  const indice = NIVELES.findIndex(n => n.id === res.nivel.id);
  const siguiente = NIVELES[indice + 1];
  const btnSig = $('#btn-siguiente');
  if(res.victoria && siguiente){
    btnSig.hidden = false;
    btnSig.dataset.nivel = siguiente.id;
    btnSig.textContent = `Siguiente: ${siguiente.nombre} ${siguiente.icono} ➡️`;
  }else{
    btnSig.hidden = true;
  }
  $('#btn-repetir').dataset.nivel = res.nivel.id;
}

function mostrarPantalla(id){
  document.querySelectorAll('.pantalla').forEach(p => {
    p.hidden = true;
    p.classList.remove('activa');
  });
  const destino = document.getElementById(id);
  destino.hidden = false;
  void destino.offsetWidth;          // reinicia la animación de entrada
  destino.classList.add('activa');
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function volverAlMenu(){
  Juego.limpiarTimeouts();
  clearInterval(Juego.cronometro);
  Juego.detenerTemporizador();
  $('#overlay-ruleta').hidden = true;

  pintarInsignia();
}

function conectarEventos(){
  // Menú
  $('#btn-jugar').addEventListener('click', () => {
    Sonido.reproducir('click');
    Juego.iniciar(nivelSeleccionado);
  });

  // Ruleta
  $('#btn-girar').addEventListener('click', () => Juego.girarRuleta());

  // Salir de la partida (con confirmación)
  $('#btn-salir').addEventListener('click', () => { $('#overlay-salir').hidden = false; });
  $('#confirmar-seguir').addEventListener('click', () => { $('#overlay-salir').hidden = true; });
  $('#confirmar-salir').addEventListener('click', () => {
    $('#overlay-salir').hidden = true;
    volverAlMenu();
  });

  // Pantalla final
  $('#btn-menu').addEventListener('click', () => { Sonido.reproducir('click'); volverAlMenu(); });
  $('#btn-repetir').addEventListener('click', e => {
    Sonido.reproducir('click');
    Juego.iniciar(e.currentTarget.dataset.nivel);
  });
  $('#btn-siguiente').addEventListener('click', e => {
    Sonido.reproducir('click');
    Juego.iniciar(e.currentTarget.dataset.nivel);
  });

  // Sonido (todos los botones con clase .btn-sonido)
  document.querySelectorAll('.btn-sonido').forEach(b =>
    b.addEventListener('click', () => {
      Progreso.alternarSonido();
      actualizarBotonesSonido();
      Sonido.reproducir('click');
    })
  );

  // Atajos de teclado: ENTER gira la ruleta, teclas 1-4 eligen pez
  document.addEventListener('keydown', e => {
    if(!$('#pantalla-juego').classList.contains('activa')) return;
    if(!$('#overlay-ruleta').hidden){
      if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); Juego.girarRuleta(); }
      return;
    }
    const indice = ['1', '2', '3', '4'].indexOf(e.key);
    if(indice !== -1){
      const peces = document.querySelectorAll('.pez');
      if(peces[indice] && !peces[indice].disabled) peces[indice].click();
    }
  });
}

document.addEventListener('DOMContentLoaded', () => {
  Progreso.cargar();
  nivelSeleccionado = Progreso.datos.nivelMaximo || 'facil';  // continúa donde lo dejó
  crearFondoVivo();
  pintarTarjetasNivel();
  pintarInsignia();
  actualizarBotonesSonido();
  conectarEventos();
  $('#anio').textContent = new Date().getFullYear();
  mostrarPantalla('pantalla-inicio');
});
